import type React from 'react'
import {useEffect, useRef, useState} from 'react'

interface SignaturePadProps {
  onSave: (signatureDataUrl: string) => void
  onCancel: () => void
}

const SignaturePad: React.FC<SignaturePadProps> = ({onSave, onCancel}) => {

  const initCanvas = () => {

        .exec((res) => {
          if (res && res[0]) {
            const canvas = res[0].node
            const ctx = canvas.getContext('2d')

            if (!ctx) {
              console.error('无法获取Canvas上下文')
              Taro.showToast({
                title: 'Canvas初始化失败',
                icon: 'none'
              })
              return
            }

            // 获取设备像素比
            const dpr = Taro.getSystemInfoSync().pixelRatio || 2
            const width = res[0].width
            const height = res[0].height

            // 设置Canvas实际尺寸（考虑设备像素比）
            canvas.width = width * dpr
            canvas.height = height * dpr

            // 缩放绘图上下文以匹配设备像素比
            ctx.scale(dpr, dpr)

            // 设置画笔样式
            ctx.strokeStyle = '#000000'
            ctx.lineWidth = 3
            ctx.lineCap = 'round'
            ctx.lineJoin = 'round'

            // 填充白色背景
            ctx.fillStyle = '#FFFFFF'
            ctx.fillRect(0, 0, width, height)

            // 保存上下文引用
            contextRef.current = {
              ctx,
              canvas,
              width,
              height,
              dpr
            }

            setCanvasReady(true)

            console.log('Canvas初始化成功', {width, height, dpr})
          } else {
            console.error('Canvas节点查询失败')
            Taro.showToast({
              title: 'Canvas初始化失败',
              icon: 'none'
            })
          }
        })
    } catch (error) {
      console.error('Canvas初始化异常:', error)
      Taro.showToast({
        title: 'Canvas初始化异常',
        icon: 'none'
      })
    }
  }

  // 开始绘制
  const handleTouchStart = (e: any) => {
    if (!contextRef.current || !canvasReady) {
      console.warn('Canvas未就绪')
      return

    e.preventDefault()
    e.stopPropagation()

    const touch = e.touches[0]
    const {x, y} = touch

    lastPointRef.current = {x, y}
    setIsDrawing(true)
    setIsEmpty(false)

    console.log('开始绘制', {x, y})

  }

  // 结束绘制
  const handleTouchEnd = () => {
    setIsDrawing(false)
    lastPointRef.current = null
    console.log('结束绘制')
  }

  // 清除签名
  const handleClear = () => {
    if (!contextRef.current || !canvasReady) {
      console.warn('Canvas未就绪')
      return
    }

    const {ctx, width, height} = contextRef.current
    ctx.fillStyle = '#FFFFFF'
    ctx.fillRect(0, 0, width, height)
    setIsEmpty(true)

    Taro.showToast({
      title: '已清除',
      icon: 'success',
      duration: 1000
    })
  }

  // 保存签名
  const handleSave = async () => {
    if (isEmpty) {
      Taro.showToast({
        title: '请先签名',
        icon: 'none',
        duration: 2000
      })
      return
    }

    if (!contextRef.current || !canvasReady) {
      Taro.showToast({
        title: 'Canvas未就绪',
        icon: 'none',
        duration: 2000
      })
      return
    }

    try {
      Taro.showLoading({title: '保存中...'})

      const {canvas} = contextRef.current

      // 使用新版API将Canvas转换为图片
      const res = await Taro.canvasToTempFilePath({
        canvas,
        fileType: 'png',
        quality: 1
      })

      Taro.hideLoading()

      if (res.tempFilePath) {
        console.log('签名保存成功:', res.tempFilePath)
        onSave(res.tempFilePath)
      } else {
        throw new Error('未获取到图片路径')
      }
    } catch (error) {
      Taro.hideLoading()
      console.error('保存签名失败:', error)
      Taro.showToast({
        title: '保存失败，请重试',
        icon: 'error',
        duration: 2000

  }

  onCancel: () => void

            if (!ctx) {
              console.error('无法获取Canvas上下文')
              Taro.showToast({
                title: 'Canvas初始化失败',
                icon: 'none'
              })
              return

            // 获取设备像素比
            const dpr = Taro.getSystemInfoSync().pixelRatio || 2
            const width = res[0].width
            const height = res[0].height

            // 设置Canvas实际尺寸（考虑设备像素比）
            canvas.width = width * dpr
            canvas.height = height * dpr

            // 缩放绘图上下文以匹配设备像素比
            ctx.scale(dpr, dpr)

            // 设置画笔样式
            ctx.strokeStyle = '#000000'
            ctx.lineWidth = 3
            ctx.lineCap = 'round'
            ctx.lineJoin = 'round'

              width,

              title: 'Canvas初始化失败',
              icon: 'none'

## 六、注意事项

### 6.1 Canvas API注意事项

1. **必须传入canvas实例**
   ```typescript
   // ✅ 正确
   await Taro.canvasToTempFilePath({
     canvas,  // 传入canvas实例
     fileType: 'png',
     quality: 1
   })

   // ❌ 错误（旧版API）
   await Taro.canvasToTempFilePath({
     canvasId,  // 只传入canvasId
   })
   ```

2. **延迟初始化**
   - Canvas初始化必须延迟，确保DOM已渲染
   - 建议延迟300ms以上

3. **设备像素比**
   - 必须考虑设备像素比，否则签名模糊
   - 使用`Taro.getSystemInfoSync().pixelRatio`

### 6.2 状态管理注意事项

1. **Canvas就绪状态**
   - 所有Canvas操作前必须检查`canvasReady`
   - 初始化失败时要有明确提示

2. **签名状态**
   - 使用`isEmpty`状态管理签名是否为空
   - 未签名时禁用保存按钮

### 6.3 错误处理注意事项

1. **完善的错误处理**
   - 所有异步操作都要try-catch
   - 错误要有详细的日志
   - 用户要有友好的提示

2. **调试日志**
   - 关键操作要有console.log
   - 便于问题排查
