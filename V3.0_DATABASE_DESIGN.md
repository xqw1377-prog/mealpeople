# 3.0版本数据库架构设计

## 📋 设计概述

### 设计原则
1. **数据完整性**：确保员工全生命周期数据的完整记录
2. **可扩展性**：支持未来功能扩展和数据增长
3. **性能优化**：合理的索引和查询优化
4. **安全隔离**：租户级别的数据隔离和权限控制
5. **数据质量**：为AI评估和求职平台提供高质量数据

### 数据库版本
- 基于版本：2.0.0（迁移编号：32）
- 目标版本：3.0.0（迁移编号：34+）
- 版本标记：33（已创建）

## 🗄️ 表结构设计

### 一、招聘管理模块

#### 1.1 职位表 (job_positions)

```sql
CREATE TABLE job_positions (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  store_id uuid REFERENCES stores(id) ON DELETE SET NULL,
  department_id uuid REFERENCES departments(id) ON DELETE SET NULL,
  
  -- 职位基本信息
  position_name text NOT NULL,
  position_code text,
  position_level text, -- 初级、中级、高级、专家
  employment_type text NOT NULL, -- 全职、兼职、实习、外包
  
  -- 薪资信息
  salary_min numeric(10,2),
  salary_max numeric(10,2),
  salary_currency text DEFAULT 'CNY',
  
  -- 招聘信息
  headcount integer NOT NULL DEFAULT 1,
  urgency_level text DEFAULT 'normal', -- low, normal, high, urgent
  status text NOT NULL DEFAULT 'open', -- open, paused, closed, filled
  
  -- 职位要求
  requirements jsonb, -- {education, experience, skills, certificates}
  responsibilities jsonb, -- 岗位职责数组
  skills_required jsonb, -- 技能要求数组
  
  -- 职位描述
  description text,
  benefits text,
  work_location text,
  work_schedule text,
  
  -- 招聘负责人
  recruiter_id uuid REFERENCES employees(id),
  hiring_manager_id uuid REFERENCES employees(id),
  
  -- 时间戳
  posted_date date,
  deadline_date date,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now(),
  closed_at timestamptz,
  
  -- 统计信息
  applications_count integer DEFAULT 0,
  interviews_count integer DEFAULT 0,
  offers_count integer DEFAULT 0,
  
  -- 索引
  CONSTRAINT valid_salary CHECK (salary_min IS NULL OR salary_max IS NULL OR salary_min <= salary_max),
  CONSTRAINT valid_headcount CHECK (headcount > 0)
);

-- 索引
CREATE INDEX idx_job_positions_tenant ON job_positions(tenant_id);
CREATE INDEX idx_job_positions_status ON job_positions(status);
CREATE INDEX idx_job_positions_posted_date ON job_positions(posted_date DESC);
CREATE INDEX idx_job_positions_store ON job_positions(store_id);

-- 注释
COMMENT ON TABLE job_positions IS '职位表 - 存储招聘职位信息';
COMMENT ON COLUMN job_positions.urgency_level IS '紧急程度：low-低, normal-正常, high-高, urgent-紧急';
COMMENT ON COLUMN job_positions.status IS '状态：open-招聘中, paused-已暂停, closed-已关闭, filled-已招满';
```

#### 1.2 应聘表 (job_applications)

```sql
CREATE TABLE job_applications (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  position_id uuid NOT NULL REFERENCES job_positions(id) ON DELETE CASCADE,
  tenant_id uuid NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  
  -- 应聘者信息
  applicant_name text NOT NULL,
  gender text,
  age integer,
  email text,
  phone text,
  
  -- 简历信息
  resume_url text,
  resume_parsed_data jsonb, -- 解析后的简历数据
  education_background jsonb, -- 教育背景数组
  work_experience jsonb, -- 工作经历数组
  skills jsonb, -- 技能数组
  certificates jsonb, -- 证书数组
  
  -- 应聘状态
  application_status text NOT NULL DEFAULT 'pending', 
  -- pending, screening, interview, offer, accepted, rejected, withdrawn
  
  -- 评分信息
  resume_score numeric(5,2), -- 简历评分 0-100
  screening_score numeric(5,2), -- 筛选评分 0-100
  overall_score numeric(5,2), -- 综合评分 0-100
  
  -- 来源信息
  source text, -- 招聘渠道：网站、推荐、猎头等
  referrer_id uuid REFERENCES employees(id), -- 推荐人
  
  -- 备注和标签
  notes text,
  tags text[], -- 标签数组
  
  -- 处理信息
  screener_id uuid REFERENCES employees(id), -- 筛选人
  screened_at timestamptz,
  screening_notes text,
  
  -- 时间戳
  applied_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now(),
  rejected_at timestamptz,
  rejection_reason text,
  
  -- 约束
  CONSTRAINT valid_age CHECK (age IS NULL OR (age >= 16 AND age <= 100)),
  CONSTRAINT valid_scores CHECK (
    (resume_score IS NULL OR (resume_score >= 0 AND resume_score <= 100)) AND
    (screening_score IS NULL OR (screening_score >= 0 AND screening_score <= 100)) AND
    (overall_score IS NULL OR (overall_score >= 0 AND overall_score <= 100))
  )
);

-- 索引
CREATE INDEX idx_applications_position ON job_applications(position_id);
CREATE INDEX idx_applications_tenant ON job_applications(tenant_id);
CREATE INDEX idx_applications_status ON job_applications(application_status);
CREATE INDEX idx_applications_applied_at ON job_applications(applied_at DESC);
CREATE INDEX idx_applications_phone ON job_applications(phone);
CREATE INDEX idx_applications_email ON job_applications(email);

-- 注释
COMMENT ON TABLE job_applications IS '应聘表 - 存储应聘者信息和简历';
COMMENT ON COLUMN job_applications.application_status IS '状态：pending-待处理, screening-筛选中, interview-面试中, offer-已发offer, accepted-已接受, rejected-已拒绝, withdrawn-已撤回';
```

#### 1.3 面试表 (interviews)

```sql
CREATE TABLE interviews (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),

  position_id uuid NOT NULL REFERENCES job_positions(id) ON DELETE CASCADE,

  -- 面试基本信息
  interview_round integer NOT NULL DEFAULT 1, -- 面试轮次
  interview_type text NOT NULL, -- phone, video, onsite, group
  interview_title text,
  
  -- 面试安排
  scheduled_time timestamptz NOT NULL,
  duration_minutes integer DEFAULT 60,
  location text,
  meeting_link text,
  
  -- 面试官信息
  interviewer_id uuid NOT NULL REFERENCES employees(id),
  additional_interviewers uuid[], -- 其他面试官ID数组
  
  -- 面试状态
  status text NOT NULL DEFAULT 'scheduled',
  -- scheduled, in_progress, completed, cancelled, no_show
  
  -- 面试评估
  evaluation_form jsonb, -- 评估表单结构
  evaluation_scores jsonb, -- 各维度评分
  overall_score numeric(5,2), -- 综合评分 0-100
  
  -- 面试反馈
  strengths text, -- 优势
  weaknesses text, -- 不足
  comments text, -- 评语
  recommendation text, -- 推荐意见：strong_hire, hire, maybe, no_hire
  
  -- 面试结果
  result text, -- pass, fail, pending
  next_round_scheduled boolean DEFAULT false,
  
  -- 面试记录
  interview_notes text,
  recording_url text, -- 面试录音/视频URL
  
  -- 时间戳
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now(),
  completed_at timestamptz,
  cancelled_at timestamptz,
  cancellation_reason text,
  
  -- 约束
  CONSTRAINT valid_round CHECK (interview_round > 0),
  CONSTRAINT valid_duration CHECK (duration_minutes > 0),
  CONSTRAINT valid_score CHECK (overall_score IS NULL OR (overall_score >= 0 AND overall_score <= 100))
);

-- 索引
CREATE INDEX idx_interviews_application ON interviews(application_id);
CREATE INDEX idx_interviews_position ON interviews(position_id);
CREATE INDEX idx_interviews_tenant ON interviews(tenant_id);
CREATE INDEX idx_interviews_interviewer ON interviews(interviewer_id);
CREATE INDEX idx_interviews_scheduled_time ON interviews(scheduled_time);
CREATE INDEX idx_interviews_status ON interviews(status);

-- 注释
COMMENT ON TABLE interviews IS '面试表 - 存储面试安排和评估记录';
COMMENT ON COLUMN interviews.interview_type IS '面试类型：phone-电话, video-视频, onsite-现场, group-群面';
COMMENT ON COLUMN interviews.status IS '状态：scheduled-已安排, in_progress-进行中, completed-已完成, cancelled-已取消, no_show-未到场';
COMMENT ON COLUMN interviews.recommendation IS '推荐意见：strong_hire-强烈推荐, hire-推荐, maybe-待定, no_hire-不推荐';
```

```sql
CREATE TABLE job_offers (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  application_id uuid NOT NULL REFERENCES job_applications(id) ON DELETE CASCADE,
  position_id uuid NOT NULL REFERENCES job_positions(id) ON DELETE CASCADE,
  tenant_id uuid NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  
  -- Offer信息
  offer_letter_url text,
  offer_date date NOT NULL,
  expiry_date date NOT NULL,
  
  -- 薪资信息
  base_salary numeric(10,2) NOT NULL,
  bonus numeric(10,2),
  stock_options numeric(10,2),
  other_benefits jsonb,
  total_compensation numeric(10,2),
  
  -- 职位信息
  position_title text NOT NULL,
  department text,
  start_date date,
  probation_period_months integer DEFAULT 3,
  
  -- Offer状态

  -- pending, accepted, rejected, expired, withdrawn
  
  -- 响应信息

  response_notes text,
  
  -- 谈判记录
```

#### 3.1 目标表 (objectives)

```sql
CREATE TABLE objectives (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),

  level text, -- company, department, team, individual

  period text NOT NULL, -- monthly, quarterly, yearly

  end_date date NOT NULL,
  
  -- 目标状态
  status text NOT NULL DEFAULT 'active',
  -- draft, active, in_progress, completed, cancelled, overdue

  approver_id uuid REFERENCES employees(id),
  approved_at timestamptz,
```

```sql
    (competency_score >= 0 AND competency_score <= 100) AND
```

#### 3.3 360度评估表 (peer_reviews)

```sql
CREATE TABLE peer_reviews (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  employee_id uuid NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  reviewer_id uuid NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  tenant_id uuid NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  
  -- 评估周期
  review_period text NOT NULL,
  period_start_date date NOT NULL,
  period_end_date date NOT NULL,
  
  -- 评估关系
  relationship_type text NOT NULL,
  -- manager, peer, subordinate, cross_functional, self
  
  -- 评估维度
  collaboration_score numeric(5,2), -- 协作能力 0-100
  communication_score numeric(5,2), -- 沟通能力 0-100
  leadership_score numeric(5,2), -- 领导力 0-100
  innovation_score numeric(5,2), -- 创新能力 0-100
  reliability_score numeric(5,2), -- 可靠性 0-100
  
  -- 综合评分
  overall_score numeric(5,2), -- 综合评分 0-100
  
  -- 评估内容
  strengths text, -- 优势
  areas_for_improvement text, -- 待改进
  specific_examples text, -- 具体事例
  comments text, -- 评语
  
  -- 匿名设置
  is_anonymous boolean DEFAULT false,
  
  -- 评估状态
  status text NOT NULL DEFAULT 'pending',
  -- pending, submitted, reviewed
  
  -- 时间戳
  created_at timestamptz DEFAULT now(),
  submitted_at timestamptz,
  
  -- 约束
  CONSTRAINT valid_period_dates CHECK (period_end_date >= period_start_date),
  CONSTRAINT valid_scores CHECK (
    (collaboration_score IS NULL OR (collaboration_score >= 0 AND collaboration_score <= 100)) AND
    (communication_score IS NULL OR (communication_score >= 0 AND communication_score <= 100)) AND
    (leadership_score IS NULL OR (leadership_score >= 0 AND leadership_score <= 100)) AND
    (innovation_score IS NULL OR (innovation_score >= 0 AND innovation_score <= 100)) AND
    (reliability_score IS NULL OR (reliability_score >= 0 AND reliability_score <= 100)) AND
    (overall_score >= 0 AND overall_score <= 100)
  ),
  CONSTRAINT no_self_review CHECK (employee_id != reviewer_id OR relationship_type = 'self')
);

-- 索引
CREATE INDEX idx_peer_reviews_employee ON peer_reviews(employee_id);
CREATE INDEX idx_peer_reviews_reviewer ON peer_reviews(reviewer_id);
CREATE INDEX idx_peer_reviews_tenant ON peer_reviews(tenant_id);
CREATE INDEX idx_peer_reviews_period ON peer_reviews(review_period, period_start_date);
CREATE INDEX idx_peer_reviews_relationship ON peer_reviews(relationship_type);

-- 注释
COMMENT ON TABLE peer_reviews IS '360度评估表 - 存储多维度评估记录';
COMMENT ON COLUMN peer_reviews.relationship_type IS '评估关系：manager-上级, peer-同事, subordinate-下属, cross_functional-跨部门, self-自评';
```
