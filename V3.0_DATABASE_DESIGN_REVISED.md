# 3.0版本数据库设计（修订版）- 员工视角

## 📋 设计理念

### 从员工视角设计数据库
- **员工是主角**：所有数据围绕员工展开
- **简单易懂**：表名和字段名通俗易懂
- **餐饮特色**：符合餐饮行业的业务逻辑
- **成长导向**：记录员工的成长轨迹

### 数据库结构概览

## 🗄️ 详细表结构

### 一、员工核心数据

#### 1.1 员工工作信息表 (employee_work_info)

```sql
/*
员工工作信息表
用途：记录员工的工作相关信息
视角：员工视角 - 我的工作信息
*/

CREATE TABLE employee_work_info (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  employee_id uuid NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  tenant_id uuid NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  
  -- 当前工作信息
  current_position text NOT NULL, -- 当前岗位：服务员、厨师、收银员等
  position_level text DEFAULT 'junior', -- 岗位等级：新手、进阶、熟练、资深、专家
  department text, -- 部门：前厅、后厨、收银等
  store_id uuid REFERENCES stores(id), -- 所属门店
  
  -- 入职信息
  hire_date date NOT NULL, -- 入职日期
  probation_end_date date, -- 试用期结束日期
  employment_type text NOT NULL DEFAULT 'full_time', -- 全职、兼职、实习
  contract_type text, -- 合同类型：劳动合同、劳务合同等
  
  -- 工作状态
  work_status text NOT NULL DEFAULT 'active', 
  -- active-在职, probation-试用期, leave-请假, resigned-已离职
  
  -- 导师信息
  mentor_id uuid REFERENCES employees(id), -- 导师（老员工带新员工）
  mentor_assigned_date date,
  
  -- 工作搭档（餐饮特色：固定搭档）
  regular_partners uuid[], -- 常规搭档ID数组
  
  -- 工作偏好
  preferred_shifts text[], -- 偏好班次：早班、中班、晚班
  preferred_days text[], -- 偏好工作日：周一到周日
  available_for_overtime boolean DEFAULT true, -- 是否接受加班
  
  -- 紧急联系人
  emergency_contact_name text,
  emergency_contact_phone text,
  emergency_contact_relationship text,
  
  -- 时间戳
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- 索引
CREATE INDEX idx_employee_work_info_employee ON employee_work_info(employee_id);
CREATE INDEX idx_employee_work_info_tenant ON employee_work_info(tenant_id);
CREATE INDEX idx_employee_work_info_store ON employee_work_info(store_id);
CREATE INDEX idx_employee_work_info_status ON employee_work_info(work_status);
CREATE INDEX idx_employee_work_info_position ON employee_work_info(current_position);

-- 注释
COMMENT ON TABLE employee_work_info IS '员工工作信息表 - 从员工视角记录工作信息';
COMMENT ON COLUMN employee_work_info.position_level IS '岗位等级：junior-新手, intermediate-进阶, proficient-熟练, senior-资深, expert-专家';
```

### 二、工作相关

#### 2.1 工作考勤表 (work_attendance)

```sql
/*
工作考勤表
用途：记录员工的考勤情况
视角：员工视角 - 我的考勤记录
*/

CREATE TABLE work_attendance (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  employee_id uuid NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  tenant_id uuid NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  store_id uuid REFERENCES stores(id),
  
  -- 考勤日期
  attendance_date date NOT NULL,
  
  -- 排班信息
  scheduled_shift text, -- 计划班次：早班、中班、晚班
  scheduled_start_time time NOT NULL,
  scheduled_end_time time NOT NULL,
  scheduled_hours numeric(4,2), -- 计划工作小时数
  
  -- 实际考勤
  actual_start_time time,
  actual_end_time time,
  actual_hours numeric(4,2), -- 实际工作小时数
  
  -- 考勤状态
  status text NOT NULL DEFAULT 'scheduled',
  -- scheduled-已排班, checked_in-已签到, working-工作中, 
  -- checked_out-已签退, completed-已完成, absent-缺勤, leave-请假
  
  -- 签到签退
  check_in_time timestamptz,
  check_in_location text, -- 签到地点
  check_out_time timestamptz,
  check_out_location text, -- 签退地点
  
  -- 迟到早退
  is_late boolean DEFAULT false,
  late_minutes integer DEFAULT 0,
  is_early_leave boolean DEFAULT false,
  early_leave_minutes integer DEFAULT 0,
  
  -- 加班
  is_overtime boolean DEFAULT false,
  overtime_hours numeric(4,2) DEFAULT 0,
  overtime_approved boolean DEFAULT false,
  
  -- 请假
  is_leave boolean DEFAULT false,
  leave_type text, -- sick-病假, personal-事假, annual-年假
  leave_reason text,
  leave_approved boolean DEFAULT false,
  
  -- 备注
  notes text,
  
  -- 时间戳
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- 索引
CREATE INDEX idx_work_attendance_employee ON work_attendance(employee_id);
CREATE INDEX idx_work_attendance_tenant ON work_attendance(tenant_id);
CREATE INDEX idx_work_attendance_date ON work_attendance(attendance_date DESC);
CREATE INDEX idx_work_attendance_status ON work_attendance(status);

-- 唯一约束：每个员工每天只有一条考勤记录
CREATE UNIQUE INDEX idx_work_attendance_unique ON work_attendance(employee_id, attendance_date);

-- 注释
COMMENT ON TABLE work_attendance IS '工作考勤表 - 员工每日考勤记录';
```

#### 2.2 工作任务表 (work_tasks)

```sql
/*
工作任务表
用途：记录员工的日常工作任务
视角：员工视角 - 我的今日任务
*/

CREATE TABLE work_tasks (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  employee_id uuid NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  tenant_id uuid NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  store_id uuid REFERENCES stores(id),
  
  -- 任务信息
  task_name text NOT NULL, -- 任务名称：开档准备、午餐高峰、清洁卫生等
  task_type text NOT NULL, -- daily-日常, special-特殊, training-培训
  task_category text, -- 任务分类：服务、清洁、准备、整理等
  description text,
  
  -- 任务时间
  task_date date NOT NULL,
  start_time time,
  end_time time,
  estimated_duration integer, -- 预计时长（分钟）
  actual_duration integer, -- 实际时长（分钟）
  
  -- 任务状态
  status text NOT NULL DEFAULT 'pending',
  -- pending-待开始, in_progress-进行中, completed-已完成, skipped-已跳过
  
  -- 完成信息
  completed_at timestamptz,
  completion_notes text, -- 完成备注
  
  -- 任务评价
  quality_score numeric(3,1), -- 质量评分 1-5
  efficiency_score numeric(3,1), -- 效率评分 1-5
  
  -- 工作数据（自动记录）
  customers_served integer, -- 服务客户数
  orders_processed integer, -- 处理订单数
  revenue_generated numeric(10,2), -- 产生营业额
  
  -- 提醒设置
  reminder_enabled boolean DEFAULT true,
  reminder_minutes integer DEFAULT 15, -- 提前15分钟提醒
  
  -- 时间戳
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- 索引
CREATE INDEX idx_work_tasks_employee ON work_tasks(employee_id);
CREATE INDEX idx_work_tasks_tenant ON work_tasks(tenant_id);
CREATE INDEX idx_work_tasks_date ON work_tasks(task_date DESC);
CREATE INDEX idx_work_tasks_status ON work_tasks(status);

-- 注释
COMMENT ON TABLE work_tasks IS '工作任务表 - 员工日常工作任务';
COMMENT ON COLUMN work_tasks.task_type IS '任务类型：daily-日常任务, special-特殊任务, training-培训任务';
```

### 三、收入相关

#### 3.1 工资记录表 (salary_records)

```sql
/*
工资记录表
用途：记录员工的工资发放情况
视角：员工视角 - 我的工资明细
*/

CREATE TABLE salary_records (
  -- 主键和外键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  employee_id uuid NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  tenant_id uuid NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  
  -- 工资周期
  salary_month text NOT NULL, -- 格式：2025-11
  pay_date date NOT NULL, -- 发薪日期
  
  -- 基本工资
  base_salary numeric(10,2) NOT NULL, -- 基本工资
  
  -- 出勤相关
  work_days integer NOT NULL, -- 出勤天数
  work_hours numeric(6,2) NOT NULL, -- 工作小时数
  hourly_rate numeric(8,2), -- 时薪
  
  -- 加班费
  overtime_hours numeric(6,2) DEFAULT 0, -- 加班小时数
  overtime_pay numeric(10,2) DEFAULT 0, -- 加班费
  
  -- 绩效奖金
  performance_bonus numeric(10,2) DEFAULT 0, -- 绩效奖金
  quality_bonus numeric(10,2) DEFAULT 0, -- 质量奖金
  attendance_bonus numeric(10,2) DEFAULT 0, -- 全勤奖
  
  -- 其他收入
  other_income numeric(10,2) DEFAULT 0, -- 其他收入
  other_income_notes text, -- 其他收入说明
  
  -- 扣除项
  deductions numeric(10,2) DEFAULT 0, -- 扣款
  deduction_notes text, -- 扣款说明
  
  -- 总计
  gross_salary numeric(10,2) NOT NULL, -- 应发工资
  net_salary numeric(10,2) NOT NULL, -- 实发工资
  
  -- 发放状态
  payment_status text NOT NULL DEFAULT 'pending',
  -- pending-待发放, paid-已发放, delayed-延迟发放
  
  -- 工资条
  payslip_url text, -- 工资条PDF链接
  
  -- 备注
  notes text,
  
  -- 时间戳
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- 索引
CREATE INDEX idx_salary_records_employee ON salary_records(employee_id);
CREATE INDEX idx_salary_records_tenant ON salary_records(tenant_id);
CREATE INDEX idx_salary_records_month ON salary_records(salary_month DESC);
CREATE INDEX idx_salary_records_pay_date ON salary_records(pay_date DESC);

-- 唯一约束：每个员工每月只有一条工资记录
CREATE UNIQUE INDEX idx_salary_records_unique ON salary_records(employee_id, salary_month);

-- 注释
COMMENT ON TABLE salary_records IS '工资记录表 - 员工每月工资明细';
```

### 四、成长相关

#### 4.1 技能标签表 (skill_tags)

```sql
/*
技能标签表
用途：定义餐饮行业的技能标签体系
视角：行业标准 - 餐饮技能分类
*/

CREATE TABLE skill_tags (
  -- 主键
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- 标签信息
  tag_name text NOT NULL, -- 标签名称：点餐服务、炒菜技能等
  tag_code text UNIQUE, -- 标签代码：service_ordering, cooking_stir_fry
  
  -- 标签分类
  category text NOT NULL, -- 分类：服务技能、厨艺技能、管理技能
  sub_category text, -- 子分类：基础服务、高级服务等
  
  -- 标签层级
  parent_tag_id uuid REFERENCES skill_tags(id),
  level integer DEFAULT 1, -- 层级：1-一级, 2-二级, 3-三级
  
  -- 标签描述
  description text,
  
  -- 适用岗位
  applicable_positions text[], -- 适用岗位数组
  
  -- 评估标准
  assessment_criteria jsonb, -- 评估标准
  level_definitions jsonb, -- 等级定义：初级、中级、高级等
  
  -- 标签权重
  importance_weight numeric(3,1) DEFAULT 5.0, -- 重要性权重 1-10
  
  -- 标签状态
  is_active boolean DEFAULT true,
  
  -- 排序
  display_order integer DEFAULT 0,
  
  -- 时间戳
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- 索引
CREATE INDEX idx_skill_tags_category ON skill_tags(category);
CREATE INDEX idx_skill_tags_parent ON skill_tags(parent_tag_id);
CREATE INDEX idx_skill_tags_active ON skill_tags(is_active) WHERE is_active = true;

-- 注释
COMMENT ON TABLE skill_tags IS '技能标签表 - 餐饮行业技能分类体系';
```

```sql
  updated_at timestamptz DEFAULT now()
);

-- 索引
CREATE INDEX idx_talent_profiles_employee ON talent_profiles(employee_id);
CREATE INDEX idx_talent_profiles_score ON talent_profiles(overall_score DESC);
CREATE INDEX idx_talent_profiles_level ON talent_profiles(talent_level);
CREATE INDEX idx_talent_profiles_status ON talent_profiles(job_seeking_status);

-- 唯一约束：每个员工只有一份人才档案
CREATE UNIQUE INDEX idx_talent_profiles_unique ON talent_profiles(employee_id);

-- 注释
COMMENT ON TABLE talent_profiles IS '人才档案表 - 为招募平台准备的人才数据';
```

## 🔒 安全策略

### RLS策略设计

#### 员工数据访问规则
```sql
-- 员工可以查看自己的所有数据
CREATE POLICY "员工查看自己的数据" ON employee_work_info
  FOR SELECT USING (auth.uid() = employee_id);

-- 员工可以更新自己的部分数据
CREATE POLICY "员工更新自己的数据" ON employee_work_info
  FOR UPDATE USING (auth.uid() = employee_id);

-- 管理者可以查看本店员工数据
CREATE POLICY "管理者查看本店员工数据" ON employee_work_info
  FOR SELECT USING (
    EXISTS (
      SELECT 1 FROM employees e
      WHERE e.id = auth.uid()
      AND e.role IN ('manager', 'admin')
      AND e.store_id = employee_work_info.store_id
    )
  );
```

#### 敏感数据保护
```sql
-- AI评估数据只有系统和管理员可以访问
CREATE POLICY "AI评估数据访问限制" ON ai_assessments
  FOR SELECT USING (
    auth.uid() = employee_id OR
    EXISTS (
      SELECT 1 FROM employees e
      WHERE e.id = auth.uid()
      AND e.role = 'admin'
    )
  );

-- 人才档案数据只有系统可以写入
CREATE POLICY "人才档案系统写入" ON talent_profiles
  FOR INSERT WITH CHECK (false); -- 只能通过后台函数写入
```
