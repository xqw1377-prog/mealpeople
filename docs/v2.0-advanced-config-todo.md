## 🔧 技术要点

### Excel导入实现

```typescript
// 安装依赖
pnpm add xlsx

// 解析Excel
import * as XLSX from 'xlsx'

const handleFileSelect = async (file: File) => {
  const data = await file.arrayBuffer()
  const workbook = XLSX.read(data)
  const worksheet = workbook.Sheets[workbook.SheetNames[0]]
  const jsonData = XLSX.utils.sheet_to_json(worksheet)
  
  // 数据验证和转换
  const records = jsonData.map((row: any) => ({
    tenant_id: currentTenant.id,
    store_id: selectedStore,
    revenue_date: formatDate(row['日期']),
    revenue_amount: parseFloat(row['营收金额']),
    customer_count: parseInt(row['客流量']) || null,
    staff_count: parseInt(row['员工数']) || null,
    working_hours: parseFloat(row['工时']) || null,
    labor_cost: parseFloat(row['人力成本']) || null,
    notes: row['备注'] || null,
    data_source: 'import',
    created_by: user.id
  }))
  
  // 批量导入
  const batchId = generateUUID()
  const result = await importRevenueHistory(records, batchId)
  
  // 显示结果
  showImportResult(result)
}
```

// 可顶岗岗位多选
const [selectedPositions, setSelectedPositions] = useState<string[]>([])

const handlePositionToggle = (position: string) => {
  setSelectedPositions(prev => 
    prev.includes(position)
      ? prev.filter(p => p !== position)
      : [...prev, position]
  )
}
