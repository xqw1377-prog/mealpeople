# 2.0版本智能排休系统设计方案

## 📋 需求概述

### 核心需求

1. **模块化权限系统**
   - 按岗位拆分功能模块
   - 模块可自由组合
   - 租户管理员可配置岗位的模块访问权限
   - 员工可通过小程序申请特定日子排休

2. **智能排休算法**
   - 系统根据多维度因素智能生成月度排休列表
   - 考虑营收预测、效能配置、排休规则等因素
   - 支持权重配置，平衡多个目标

3. **排休权重配置**
   - **人效权重**：通过营收及效能配置确定用工需求
   - **员工权重**：员工希望的特定休息时间
   - **体验保障权重**：核心岗位视角，保障服务质量
   - **钱效权重**：每1000元工资贡献的视角

## 🧮 智能排休算法设计

### 1. 算法流程

```
1. 数据收集阶段
   ├── 收集员工排休申请
   ├── 获取排休规则配置
   ├── 获取权重配置
   ├── 获取营收预测数据
   ├── 获取效能标准配置
   ├── 获取核心岗位配置
   └── 获取历史排休数据

2. 约束条件构建阶段
   ├── 硬约束（必须满足）
   │   ├── 每月休息天数限制
   │   ├── 不可排休日期
   │   ├── 核心岗位不可同休
   │   ├── 最大连休天数
   │   └── 最低在岗人数
   └── 软约束（尽量满足）
       ├── 员工申请优先级
       ├── 休息日均匀分布
       ├── 避免周末休息
       └── 成本优化目标

3. 初始方案生成阶段
   ├── 使用贪心算法生成初始方案
   ├── 优先满足高优先级申请
   ├── 确保硬约束满足
   └── 生成可行解

4. 方案优化阶段
   ├── 使用遗传算法或模拟退火算法
   ├── 迭代优化方案
   ├── 计算每个方案的综合得分
   └── 选择最优方案

5. 冲突检测和调整阶段
   ├── 检测所有冲突
   ├── 自动调整冲突
   ├── 生成调整建议
   └── 记录无法解决的冲突

6. 结果评估阶段
   ├── 计算各维度得分
   ├── 生成统计报告
   ├── 生成优化建议
   └── 保存排休计划
```

### 2. 评分算法

```typescript
// 综合得分计算
function calculateTotalScore(
  schedule: MonthlyRestScheduleData,
  config: RestWeightConfig
): ScoreResult {
  // 计算各维度得分
  const efficiencyScore = calculateEfficiencyScore(schedule, config)
  const employeeScore = calculateEmployeeScore(schedule, config)
  const experienceScore = calculateExperienceScore(schedule, config)
  const costScore = calculateCostScore(schedule, config)
  
  // 加权计算总分
  const totalScore = 
    efficiencyScore * (config.efficiency_weight / 100) +
    employeeScore * (config.employee_weight / 100) +
    experienceScore * (config.experience_weight / 100) +
    costScore * (config.cost_weight / 100)
  
  return {
    total_score: totalScore,
    efficiency_score: efficiencyScore,
    employee_score: employeeScore,
    experience_score: experienceScore,
    cost_score: costScore
  }
}

// 人效得分计算
function calculateEfficiencyScore(
  schedule: MonthlyRestScheduleData,
  config: RestWeightConfig
): number {
  let score = 0
  const effConfig = config.efficiency_config
  
  for (const day of schedule.daily_summary) {
    // 计算当日人效
    const dailyEfficiency = day.predicted_revenue / day.working_count
    
    // 与标准人效对比
    const standardEfficiency = getStandardEfficiency(day.date)
    const efficiencyRatio = dailyEfficiency / standardEfficiency
    
    // 在容忍范围内得满分，超出范围扣分
    if (efficiencyRatio >= 1 - effConfig.tolerance_percentage / 100 &&
        efficiencyRatio <= 1 + effConfig.tolerance_percentage / 100) {
      score += 100
    } else {
      score += Math.max(0, 100 - Math.abs(efficiencyRatio - 1) * 100)
    }
  }
  
  return score / schedule.daily_summary.length
}

// 员工满意度得分计算
function calculateEmployeeScore(
  schedule: MonthlyRestScheduleData,
  config: RestWeightConfig
): number {
  let totalScore = 0
  const empConfig = config.employee_config
  
  for (const emp of schedule.employees) {
    let empScore = 0
    
    // 申请满足率得分
    const satisfactionScore = emp.satisfaction_rate * 100
    empScore += satisfactionScore * empConfig.request_priority_weight
    
    // 工龄得分
    if (empConfig.consider_seniority) {
      const seniorityScore = calculateSeniorityScore(emp.employee_id)
      empScore += seniorityScore * empConfig.seniority_weight
    }
    
    // 绩效得分
    if (empConfig.consider_performance) {
      const performanceScore = calculatePerformanceScore(emp.employee_id)
      empScore += performanceScore * empConfig.performance_weight
    }
    
    totalScore += empScore
  }
  
  return totalScore / schedule.employees.length
}

// 体验保障得分计算
function calculateExperienceScore(
  schedule: MonthlyRestScheduleData,
  config: RestWeightConfig
): number {
  let score = 0
  const expConfig = config.experience_config
  
  for (const day of schedule.daily_summary) {
    let dayScore = 100
    
    // 检查核心岗位在岗人数
    for (const [position, minCount] of Object.entries(expConfig.core_position_min_count)) {
      const actualCount = day.position_distribution[position] || 0
      if (actualCount < minCount) {
        dayScore -= (minCount - actualCount) * 20 // 每少一人扣20分
      }
    }
    
    // 检查服务质量
    if (day.risk_level === 'high') {
      dayScore -= 30
    } else if (day.risk_level === 'medium') {
      dayScore -= 15
    }
    
    score += Math.max(0, dayScore)
  }
  
  return score / schedule.daily_summary.length
}

// 钱效得分计算
function calculateCostScore(
  schedule: MonthlyRestScheduleData,
  config: RestWeightConfig
): number {
  let score = 0
  const costConfig = config.cost_config
  
  for (const day of schedule.daily_summary) {
    // 计算当日人力成本
    const dailyCost = calculateDailyCost(day.working_employees)
    
    // 计算钱效（每1000元工资贡献的营收）
    const costEfficiency = (day.predicted_revenue / dailyCost) * 1000
    
    // 与目标钱效对比
    const targetEfficiency = costConfig.min_revenue_per_1000_salary
    const efficiencyRatio = costEfficiency / targetEfficiency
    
    // 达到或超过目标得满分
    if (efficiencyRatio >= 1) {
      score += 100
    } else {
      score += efficiencyRatio * 100
    }
  }
  
  return score / schedule.daily_summary.length
}
```

// 遗传算法优化排休方案
function optimizeScheduleWithGA(
  initialSchedule: MonthlyRestScheduleData,
  config: RestWeightConfig,
  constraints: RestConstraints
): MonthlyRestScheduleData {
  const POPULATION_SIZE = 50
  const GENERATIONS = 100
  const MUTATION_RATE = 0.1
  
  // 初始化种群
  let population = initializePopulation(initialSchedule, POPULATION_SIZE)
  
  for (let gen = 0; gen < GENERATIONS; gen++) {
    // 评估适应度
    const fitness = population.map(schedule => 
      calculateTotalScore(schedule, config).total_score
    )
    
    // 选择
    const selected = selection(population, fitness)
    
    // 交叉
    const offspring = crossover(selected)
    
    // 变异
    const mutated = mutation(offspring, MUTATION_RATE)
    
    // 更新种群
    population = mutated
    
    // 检查是否满足约束
    population = population.filter(schedule => 
      validateConstraints(schedule, constraints)
    )
  }
  
  // 返回最优解
  const finalFitness = population.map(schedule => 
    calculateTotalScore(schedule, config).total_score
  )
  const bestIndex = finalFitness.indexOf(Math.max(...finalFitness))
  
  return population[bestIndex]
}
