# 2.0版本经营区域管理系统设计方案

## 📋 需求概述

### 核心需求

1. **经营区域岗位配置系统**
   - 实现"定岗、定编、定人"的完整配置体系
   - 经营区域 → 岗位编制 → 人员分配的层级管理
   - 支持餐厅关闭/开启经营区域
   - 支持每日动态调整经营区域状态

2. **上岗一览图**
   - 排休完成后展示每个经营单元的上岗情况
   - 可视化展示各经营区域的人员配置
   - 实时显示在岗/休息状态
   - 支持多维度查看（按日期、按区域、按岗位）

---

## 🏗️ 系统架构设计

### 1. 经营区域管理系统

#### 1.1 核心概念

```
餐厅
├── 经营区域（Business Area）
│   ├── 大厅区
│   ├── 包间区
│   ├── 吧台区
│   ├── 后厨区
│   └── 外卖区
│
├── 定岗（Position Definition）
│   └── 确定每个经营区域需要哪些岗位
│       ├── 大厅区：服务员、领班
│       ├── 包间区：服务员、领班
│       ├── 吧台区：调酒师、收银员
│       ├── 后厨区：厨师长、厨师、配菜员
│       └── 外卖区：打包员、配送员
│
├── 定编（Staff Quota）
│   └── 确定每个岗位需要多少人（编制数）
│       ├── 大厅区服务员：8人
│       ├── 大厅区领班：2人
│       ├── 后厨厨师：5人
│       └── ...
│
└── 定人（Staff Assignment）
    └── 确定具体哪些员工分配到这些岗位
        ├── 大厅区服务员：张三、李四、王五...
        ├── 后厨厨师：赵六、钱七...
        └── ...
```

#### 1.2 数据库表设计

```sql
-- ==================== 1. 经营区域表 ====================

CREATE TABLE business_areas (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL,
  store_id uuid NOT NULL,
  
  -- 区域信息
  area_code text NOT NULL,              -- 区域代码（唯一标识）
  area_name text NOT NULL,              -- 区域名称
  area_type text NOT NULL,              -- 区域类型：dining/kitchen/bar/takeout/other
  description text,                     -- 区域描述
  
  -- 区域配置
  capacity int,                         -- 容纳人数/桌数
  floor_number int,                     -- 楼层
  sort_order int DEFAULT 0,             -- 排序
  
  -- 状态
  is_active boolean DEFAULT true,       -- 是否启用
  can_close_daily boolean DEFAULT true, -- 是否支持每日关闭
  
  -- 营收配置
  revenue_weight decimal(5,2) DEFAULT 1.0, -- 营收权重
  
  created_by uuid,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now(),
  
  UNIQUE(tenant_id, store_id, area_code)
);

COMMENT ON TABLE business_areas IS '经营区域表';
COMMENT ON COLUMN business_areas.area_code IS '区域代码';
COMMENT ON COLUMN business_areas.area_type IS '区域类型';
COMMENT ON COLUMN business_areas.can_close_daily IS '是否支持每日关闭';

-- ==================== 2. 经营区域每日状态表 ====================

CREATE TABLE area_daily_status (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL,
  store_id uuid NOT NULL,
  area_id uuid NOT NULL REFERENCES business_areas(id) ON DELETE CASCADE,
  
  -- 日期
  status_date date NOT NULL,
  
  -- 状态
  is_open boolean DEFAULT true,         -- 是否营业
  open_time time,                       -- 开始营业时间
  close_time time,                      -- 结束营业时间
  
  -- 原因
  close_reason text,                    -- 关闭原因
  
  -- 预测数据
  predicted_customer_count int,         -- 预测客流量
  predicted_revenue decimal(10,2),      -- 预测营收
  
  created_by uuid,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now(),
  
  UNIQUE(tenant_id, store_id, area_id, status_date)
);

COMMENT ON TABLE area_daily_status IS '经营区域每日状态表';
COMMENT ON COLUMN area_daily_status.is_open IS '是否营业';
COMMENT ON COLUMN area_daily_status.close_reason IS '关闭原因';

-- ==================== 3. 经营区域岗位配置表（定岗+定编）====================

CREATE TABLE area_positions (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL,
  store_id uuid NOT NULL,
  area_id uuid NOT NULL REFERENCES business_areas(id) ON DELETE CASCADE,
  
  -- 岗位信息（定岗）
  position_name text NOT NULL,          -- 岗位名称
  position_level text,                  -- 岗位级别：junior/intermediate/senior/manager
  
  -- 编制信息（定编）
  quota_count int NOT NULL DEFAULT 1,   -- 编制人数
  min_count int NOT NULL DEFAULT 1,     -- 最少人数
  max_count int,                        -- 最多人数
  
  -- 岗位要求
  required_skills text[],               -- 必需技能
  work_hours_per_day decimal(4,2),      -- 每日工作时长
  
  -- 薪资范围
  salary_min decimal(10,2),             -- 最低薪资
  salary_max decimal(10,2),             -- 最高薪资
  
  -- 状态
  is_active boolean DEFAULT true,
  
  created_by uuid,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now(),
  
  UNIQUE(tenant_id, store_id, area_id, position_name)
);

COMMENT ON TABLE area_positions IS '经营区域岗位配置表（定岗+定编）';
COMMENT ON COLUMN area_positions.position_name IS '岗位名称';
COMMENT ON COLUMN area_positions.quota_count IS '编制人数';
COMMENT ON COLUMN area_positions.min_count IS '最少人数';

-- ==================== 4. 经营区域人员分配表（定人）====================

CREATE TABLE area_staff_assignments (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL,
  store_id uuid NOT NULL,
  area_id uuid NOT NULL REFERENCES business_areas(id) ON DELETE CASCADE,
  area_position_id uuid NOT NULL REFERENCES area_positions(id) ON DELETE CASCADE,
  employee_id uuid NOT NULL,
  
  -- 分配信息
  assignment_type text DEFAULT 'permanent', -- permanent/temporary/backup
  start_date date NOT NULL,
  end_date date,
  
  -- 工作安排
  work_schedule jsonb,                  -- 工作时间安排
  priority int DEFAULT 0,               -- 优先级（用于顶岗排序）
  
  -- 状态
  is_active boolean DEFAULT true,
  
  created_by uuid,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now(),
  
  UNIQUE(tenant_id, store_id, area_position_id, employee_id, start_date)
);

COMMENT ON TABLE area_staff_assignments IS '经营区域人员分配表（定人）';
COMMENT ON COLUMN area_staff_assignments.assignment_type IS '分配类型：permanent固定/temporary临时/backup顶岗';
COMMENT ON COLUMN area_staff_assignments.priority IS '优先级';

-- ==================== 5. 上岗一览表 ====================

CREATE TABLE area_attendance_overview (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid NOT NULL,
  store_id uuid NOT NULL,
  overview_date date NOT NULL,
  
  -- 汇总数据
  overview_data jsonb NOT NULL,         -- 详细上岗数据
  
  -- 统计信息
  total_areas int,                      -- 总区域数
  open_areas int,                       -- 营业区域数
  total_positions int,                  -- 总岗位数
  total_staff_required int,             -- 总需求人数
  total_staff_assigned int,             -- 总分配人数
  total_staff_on_duty int,              -- 总在岗人数
  total_staff_on_rest int,              -- 总休息人数
  
  -- 状态
  is_complete boolean DEFAULT false,    -- 是否完整
  
  generated_at timestamptz DEFAULT now(),
  
  UNIQUE(tenant_id, store_id, overview_date)
);

COMMENT ON TABLE area_attendance_overview IS '上岗一览表';
COMMENT ON COLUMN area_attendance_overview.overview_data IS '详细上岗数据';
COMMENT ON COLUMN area_attendance_overview.is_complete IS '是否完整（所有岗位都有人）';

-- 添加索引
CREATE INDEX idx_business_areas_tenant ON business_areas(tenant_id);
CREATE INDEX idx_business_areas_store ON business_areas(store_id);
CREATE INDEX idx_area_daily_status_date ON area_daily_status(status_date);
CREATE INDEX idx_area_positions_area ON area_positions(area_id);
CREATE INDEX idx_area_assignments_employee ON area_staff_assignments(employee_id);
CREATE INDEX idx_area_assignments_area ON area_staff_assignments(area_id);
CREATE INDEX idx_attendance_overview_date ON area_attendance_overview(overview_date);
```

/**
 * 经营区域类型

export type AreaType = 'dining' | 'kitchen' | 'bar' | 'takeout' | 'other'

export interface BusinessArea {
  id: string
  tenant_id: string
  store_id: string
  area_code: string
  area_name: string
  area_type: AreaType

  capacity?: number
  floor_number?: number
  sort_order: number
  is_active: boolean
  can_close_daily: boolean
  revenue_weight: number

  updated_at: string

 * 经营区域每日状态

export interface AreaDailyStatus {
  id: string
  tenant_id: string
  store_id: string

  status_date: string // YYYY-MM-DD
  is_open: boolean
  open_time?: string // HH:mm
  close_time?: string // HH:mm
  close_reason?: string
  predicted_customer_count?: number
  predicted_revenue?: number
  created_by?: string
  created_at: string
  updated_at: string
}

/**
 * 岗位级别

export type PositionLevel = 'junior' | 'intermediate' | 'senior' | 'manager'

 */
export interface AreaPosition {
  id: string
  tenant_id: string
  store_id: string
  area_id: string
  position_name: string
  position_level?: PositionLevel
  quota_count: number // 编制人数
  min_count: number // 最少人数
  max_count?: number // 最多人数
  required_skills?: string[]
  work_hours_per_day?: number
  salary_min?: number
  salary_max?: number
  is_active: boolean
  created_by?: string
  created_at: string
  updated_at: string
}

/**
 * 分配类型
 */
export type AssignmentType = 'permanent' | 'temporary' | 'backup'

/**
 * 经营区域人员分配（定人）
 */
export interface AreaStaffAssignment {
  id: string
  tenant_id: string
  store_id: string
  area_id: string
  area_position_id: string
  employee_id: string
  assignment_type: AssignmentType
  start_date: string // YYYY-MM-DD
  end_date?: string // YYYY-MM-DD
  work_schedule?: any // 工作时间安排
  priority: number
  is_active: boolean
  created_by?: string
  created_at: string
  updated_at: string
}

/**
 * 上岗一览数据
 */
export interface AreaAttendanceOverview {
  id: string
  tenant_id: string
  store_id: string
  overview_date: string // YYYY-MM-DD
  overview_data: AreaAttendanceData
  total_areas: number
  open_areas: number
  total_positions: number
  total_staff_required: number
  total_staff_assigned: number
  total_staff_on_duty: number
  total_staff_on_rest: number
  is_complete: boolean
  generated_at: string
}

/**
 * 上岗一览详细数据
 */
export interface AreaAttendanceData {
  date: string
  areas: AreaAttendanceDetail[]
  summary: AttendanceSummary
}

/**
 * 区域上岗详情
 */
export interface AreaAttendanceDetail {
  area_id: string
  area_name: string
  area_type: AreaType
  is_open: boolean
  close_reason?: string
  positions: PositionAttendanceDetail[]
  area_summary: {
    total_positions: number
    total_required: number
    total_assigned: number
    total_on_duty: number
    total_on_rest: number
    fill_rate: number // 配置率
  }
}

/**
 * 岗位上岗详情
 */
export interface PositionAttendanceDetail {
  position_id: string
  position_name: string
  position_level?: PositionLevel
  quota_count: number
  min_count: number
  assigned_staff: StaffAttendanceInfo[]
  on_duty_count: number
  on_rest_count: number
  shortage_count: number // 缺员数

/**
 * 员工上岗信息
 */
export interface StaffAttendanceInfo {
  employee_id: string
  employee_name: string
  assignment_type: AssignmentType
  is_on_duty: boolean // 是否在岗
  is_on_rest: boolean // 是否休息
  rest_reason?: string // 休息原因

/**
 * 上岗汇总

export interface AttendanceSummary {

  open_areas: number
  closed_areas: number
  total_positions: number
  total_required: number
  total_assigned: number
  total_on_duty: number
  total_on_rest: number

/**
 * 缺员岗位
 */
export interface ShortagePosition {
  area_name: string
  position_name: string
  required: number
  assigned: number
  shortage: number
}

## 🔄 业务流程

### 1. 经营区域配置流程

```
1. 创建经营区域
   ├─ 定义区域基本信息
   ├─ 设置区域类型和容量
   └─ 配置营收权重

2. 定岗（配置岗位）
   ├─ 为每个区域添加所需岗位
   ├─ 设置岗位级别
   └─ 定义岗位要求

3. 定编（设置编制）
   ├─ 设置每个岗位的编制人数
   ├─ 设置最少/最多人数
   └─ 配置薪资范围

4. 定人（分配人员）
   ├─ 选择员工分配到岗位
   ├─ 设置分配类型（固定/临时/顶岗）
   ├─ 设置分配时间
   └─ 设置优先级

5. 每日状态配置
   ├─ 设置区域开启/关闭
   ├─ 配置营业时间
   └─ 填写关闭原因
```

### 2. 上岗一览生成流程

```
1. 数据收集
   ├─ 获取所有经营区域
   ├─ 获取当日区域状态
   ├─ 获取所有岗位配置
   ├─ 获取人员分配信息
   └─ 获取排休计划

2. 数据处理
   ├─ 计算每个岗位的在岗人数
   ├─ 计算每个岗位的休息人数
   ├─ 计算缺员数
   └─ 计算配置率

3. 数据汇总
   ├─ 按区域汇总
   ├─ 按岗位汇总
   ├─ 计算总体统计
   └─ 识别缺员岗位

4. 生成一览图
   ├─ 生成可视化数据
   ├─ 保存到数据库
   └─ 展示给用户
```
