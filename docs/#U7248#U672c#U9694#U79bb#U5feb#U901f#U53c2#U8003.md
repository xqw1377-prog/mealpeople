```bash
# 1.0版本迁移
# 文件位置: supabase/migrations/v1/*.sql

# 2.0版本迁移
# 文件位置: supabase/migrations/v2/*.sql

# 共享迁移（审计系统等）
# 文件位置: supabase/migrations/shared/*.sql
```

### 创建数据快照

```sql
-- 创建手动快照
SELECT create_data_snapshot(
    'backup-20250106',           -- 快照名称
    '1.0.0',                     -- 应用版本
    'manual',                    -- 快照类型
    ARRAY['tenants', 'profiles', 'employees', 'schedules', 'cost_data'],  -- 表名列表
    '<user_id>'                  -- 用户ID
);

-- 查看所有快照
SELECT * FROM data_snapshots ORDER BY created_at DESC;
```

### 查询审计日志

```sql
-- 查看最近的审计日志
SELECT * FROM audit_logs 
WHERE app_version = '1.0.0' 
ORDER BY created_at DESC 
LIMIT 100;

-- 查看特定表的操作记录
SELECT * FROM audit_logs 
WHERE table_name = 'tenants' 
AND operation_type = 'UPDATE'
ORDER BY created_at DESC;

-- 查看审计统计
SELECT * FROM audit_logs_summary 
WHERE app_version = '1.0.0'
ORDER BY operation_date DESC;
```

### 数据恢复

```sql
-- 从快照恢复数据（谨慎使用！）
SELECT restore_from_snapshot(
    '<snapshot_id>',  -- 快照ID
    '<user_id>'       -- 操作人ID
);
```

## 🔧 开发工作流

### 1.0版本维护（Bug修复）

```bash
# 1. 切换到1.0分支
git checkout release/v1.0

# 2. 创建修复分支
git checkout -b hotfix/fix-xxx

# 3. 修复bug并测试
# ... 修改代码 ...
pnpm run lint
pnpm run build:weapp

# 4. 提交更改
git add .
git commit -m "fix: 修复XXX问题"

# 5. 合并到1.0分支
git checkout release/v1.0
git merge hotfix/fix-xxx

# 6. 创建补丁版本标签
git tag -a v1.0.1 -m "Fix: 修复XXX问题"

# 7. 推送到远程
git push origin release/v1.0
git push origin v1.0.1

# 8. 删除修复分支
git branch -d hotfix/fix-xxx
```

### 2.0版本开发（新功能）

```bash
# 1. 切换到2.0分支
git checkout develop/v2.0

# 2. 创建功能分支
git checkout -b feature/new-feature

# 3. 开发新功能
# ... 修改代码 ...
pnpm run lint
pnpm run build:weapp

# 4. 提交更改
git add .
git commit -m "feat: 添加XXX功能"

# 5. 合并到2.0分支
git checkout develop/v2.0
git merge feature/new-feature

# 6. 推送到远程
git push origin develop/v2.0

# 7. 删除功能分支
git branch -d feature/new-feature
```

## 🆘 紧急处理流程

### 生产环境严重故障

```bash
# 1. 立即回滚（5分钟内）
git checkout release/v1.0
pnpm run build:weapp
# 重新部署

# 2. 通知用户（10分钟内）
# 发布公告说明情况

# 3. 数据检查（30分钟内）
# 查看审计日志
# 检查数据完整性
# 评估影响范围

# 4. 数据修复（如需要）
# 使用快照恢复
# 或手动修正数据

# 5. 问题分析（2小时内）
# 定位根本原因
# 制定修复方案

# 6. 修复验证（4小时内）
# 在测试环境修复
# 充分测试验证

# 7. 重新部署（8小时内）
# 确认无误后重新部署
# 持续监控24小时
```

### 数据异常处理

```bash
# 1. 停止相关操作
# 暂停受影响的功能

# 2. 查看审计日志
SELECT * FROM audit_logs 
WHERE table_name = '<affected_table>'
AND created_at >= '<incident_time>'
ORDER BY created_at DESC;

# 3. 评估影响范围
# 确定受影响的数据量
# 确定受影响的用户

# 4. 数据恢复
# 方案A: 使用快照恢复
SELECT restore_from_snapshot('<snapshot_id>', '<user_id>');

# 方案B: 根据审计日志手动修正
# 查看old_data，恢复原始值

# 5. 验证修复
# 检查数据完整性
# 测试相关功能

# 6. 恢复服务
# 重新启用功能
# 通知用户
```
