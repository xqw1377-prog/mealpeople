# 排班规划功能合并优化方案

## 优化目标

将"营业调整"功能合并到"排班规划"页面中，简化操作流程，提升用户体验。

## 当前架构分析

### 现有页面结构
1. **排班规划页面** (`src/pages/schedule-planning/index.tsx`)
   - 第一步：输入预估营收
   - 第二步：查表获取目标人数
   - 第三步：输入排班人数（正式工+兼职）
   - 第四步：快速排休和增加兼职

2. **排班日志页面** (`src/pages/schedule-logs/index.tsx`)
   - 展示排班记录
   - 包含：排班规划、营业调整、营业复盘三个阶段的数据

### 数据表结构

#### daily_operations 表
```sql
CREATE TABLE daily_operations (
  id UUID PRIMARY KEY,
  tenant_id UUID NOT NULL,
  store_id UUID NOT NULL,
  operation_date DATE NOT NULL,
  
  -- 排班规划阶段
  estimated_revenue DECIMAL(10,2),
  planned_work_hours DECIMAL(10,2),
  
  -- 营业中调整阶段
  midday_estimated_revenue DECIMAL(10,2),
  adjusted_work_hours DECIMAL(10,2),
  
  -- 营业后复盘阶段
  actual_revenue DECIMAL(10,2),
  actual_work_hours DECIMAL(10,2),
  per_capita_revenue DECIMAL(10,2),
  efficiency_rating TEXT,
  
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
```

#### schedule_results 表
- 存储排班结果
- 包含人数、工时、成本等详细信息

## 优化方案

### 方案一：在排班规划页面添加"营业调整"区域

#### 1. 页面布局调整

```
排班规划页面
├── 第一步：预估营收（保持不变）
├── 第二步：查表结果（保持不变）
├── 第三步：排班人数（保持不变）
├── 第四步：快速排休和增加兼职（保持不变）
└── 【新增】第五步：营业调整（可选）
    ├── 调整时间选择（早餐后/午餐后/晚餐后）
    ├── 调整后预估营收
    ├── 调整后排班人数
    ├── 调整原因说明
    └── 保存调整按钮
```

#### 2. 功能特点

**显示逻辑**：
- 只有在完成第四步（保存排班规划）后，才显示第五步
- 第五步为可选项，用户可以选择是否进行营业调整
- 支持多次调整，每次调整都会记录

**数据保存**：
- 每次调整都更新 `daily_operations` 表的调整字段
- 同时创建一条调整记录到 `schedule_logs` 表
- 记录调整时间、调整人、调整原因

```sql
-- 营业调整记录表
CREATE TABLE IF NOT EXISTS operation_adjustments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  store_id UUID NOT NULL REFERENCES stores(id) ON DELETE CASCADE,
  operation_date DATE NOT NULL,
  
  -- 调整时间
  adjustment_time TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  adjustment_period TEXT NOT NULL, -- 'after_breakfast', 'after_lunch', 'after_dinner'
  
  -- 调整前数据
  before_estimated_revenue DECIMAL(10,2),
  before_staff_count INTEGER,
  
  -- 调整后数据
  after_estimated_revenue DECIMAL(10,2),
  after_staff_count INTEGER,
  
  -- 调整信息
  adjustment_reason TEXT,
  adjusted_by UUID REFERENCES profiles(id),
  adjusted_by_name TEXT,
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  
  FOREIGN KEY (tenant_id, store_id, operation_date) 
    REFERENCES daily_operations(tenant_id, store_id, operation_date)
);

-- 启用RLS
ALTER TABLE operation_adjustments ENABLE ROW LEVEL SECURITY;

-- RLS策略
CREATE POLICY "用户可以查看自己租户的调整记录" ON operation_adjustments
  FOR SELECT USING (

      SELECT tenant_id FROM profiles WHERE id = auth.uid()

CREATE POLICY "用户可以创建自己租户的调整记录" ON operation_adjustments
  FOR INSERT WITH CHECK (
    tenant_id IN (
      SELECT tenant_id FROM profiles WHERE id = auth.uid()
    )
  );
```

```typescript
// src/db/api-operation-adjustments.ts

export interface OperationAdjustment {
  id: string
  tenant_id: string
  store_id: string
  operation_date: string
  adjustment_time: string
  adjustment_period: 'after_breakfast' | 'after_lunch' | 'after_dinner'
  before_estimated_revenue: number
  before_staff_count: number
  after_estimated_revenue: number
  after_staff_count: number
  adjustment_reason: string
  adjusted_by: string
  adjusted_by_name: string
  created_at: string
}

// 创建调整记录
export async function createOperationAdjustment(
  adjustment: Omit<OperationAdjustment, 'id' | 'created_at'>
): Promise<OperationAdjustment | null> {
  const {data, error} = await supabase
    .from('operation_adjustments')
    .insert(adjustment)
    .select()
    .single()

  if (error) {
    console.error('创建调整记录失败:', error)
    return null
  }

  return data
}

// 获取某日的调整记录
export async function getAdjustmentsByDate(
  tenantId: string,
  storeId: string,
  date: string
): Promise<OperationAdjustment[]> {
  const {data, error} = await supabase
    .from('operation_adjustments')
    .select('*')
    .eq('tenant_id', tenantId)
    .eq('store_id', storeId)
    .eq('operation_date', date)
    .order('adjustment_time', {ascending: false})

  if (error) {
    console.error('获取调整记录失败:', error)
    return []
  }

  return data || []
}
```

```typescript
// src/pages/schedule-planning/index.tsx

// 添加状态
const [showAdjustmentSection, setShowAdjustmentSection] = useState(false)

const [adjustedStaffCount, setAdjustedStaffCount] = useState('')

const [showHistory, setShowHistory] = useState(false)

// 加载调整历史
const loadAdjustmentHistory = useCallback(async () => {
  if (!currentTenant?.id || !currentStore?.id || !selectedDate) return
  
  const history = await getAdjustmentsByDate(
    currentTenant.id,
    currentStore.id,
    selectedDate
  )
  setAdjustmentHistory(history)
}, [currentTenant?.id, currentStore?.id, selectedDate])

// 保存调整
const handleSaveAdjustment = async () => {
  if (!currentTenant?.id || !currentStore?.id || !user?.id) {
    Taro.showToast({title: '缺少必要信息', icon: 'none'})
    return
  }

  if (!adjustedRevenue || !adjustedStaffCount) {
    Taro.showToast({title: '请填写完整的调整信息', icon: 'none'})
    return
  }

  if (!adjustmentReason.trim()) {
    Taro.showToast({title: '请填写调整原因', icon: 'none'})
    return
  }

  try {
    // 获取当前数据
    const operation = await getDailyOperation(
      currentTenant.id,
      currentStore.id,
      selectedDate
    )

    if (!operation) {
      Taro.showToast({title: '请先完成排班规划', icon: 'none'})
      return
    }

    // 创建调整记录
    const adjustment = await createOperationAdjustment({
      tenant_id: currentTenant.id,
      store_id: currentStore.id,
      operation_date: selectedDate,
      adjustment_period: adjustmentPeriod,
      before_estimated_revenue: operation.estimated_revenue || 0,
      before_staff_count: operation.planned_staff_count || 0,
      after_estimated_revenue: Number(adjustedRevenue),
      after_staff_count: Number(adjustedStaffCount),
      adjustment_reason: adjustmentReason,
      adjusted_by: user.id,
      adjusted_by_name: user.name || '未知'
    })

    if (adjustment) {
      // 更新 daily_operations 表
      await upsertDailyOperation({
        ...operation,
        midday_estimated_revenue: Number(adjustedRevenue),
        adjusted_work_hours: Number(adjustedStaffCount) * 8 // 假设每人8小时
      })

      Taro.showToast({title: '调整保存成功', icon: 'success'})
      
      // 重新加载历史记录
      loadAdjustmentHistory()
      
      // 清空表单
      setAdjustedRevenue('')
      setAdjustedStaffCount('')
      setAdjustmentReason('')
    }
  } catch (error) {
    console.error('保存调整失败:', error)
    Taro.showToast({title: '保存失败，请重试', icon: 'none'})
  }
}
```

```typescript
}, [currentTenant?.id, currentStore?.id, selectedDate])
```
